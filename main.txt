        // =========================================================================
        // 1. AGENT ANALYTICS ENGINE (Core Mathematical & Decision Functions)
        // =========================================================================
        function analyzeProduct(product) {
            const sum7d = product.salesHistory.reduce((acc, v) => acc + v, 0);
            const avgDailyDemand = +(sum7d / product.salesHistory.length).toFixed(1);
            const forecastDemand = Math.round(avgDailyDemand * forecastHorizonDays);

            // Identify suppliers by role
            const sortedBySpeed = [...product.suppliers].sort((a, b) => a.deliveryDays - b.deliveryDays || a.unitCost - b.unitCost);
            const sortedByCost = [...product.suppliers].sort((a, b) => a.unitCost - b.unitCost || a.deliveryDays - b.deliveryDays);

            const fastestSupplier = sortedBySpeed[0];
            const cheapestSupplier = sortedByCost[0];
            const stdLeadDays = product.primaryLeadDays || fastestSupplier.deliveryDays;

            // Expected demand during primary supplier delivery period
            const expectedDemandStdLead = Math.round(avgDailyDemand * stdLeadDays);
            const daysUntilStockout = avgDailyDemand > 0 ? +(product.stock / avgDailyDemand).toFixed(1) : 999;
            const safetyBufferUnits = Math.round(avgDailyDemand * safetyBufferDays);

            // Shortfall against standard lead time
            const stdShortfall = Math.max(0, expectedDemandStdLead - product.stock);

            // Determine Risk Classification
            let riskLevel = "LOW";
            let riskPriority = 4; // 1 = highest urgency
            let riskBadgeClass = "bg-emerald-500/15 text-emerald-300 border-emerald-500/30";

            if (daysUntilStockout < fastestSupplier.deliveryDays) {
                riskLevel = "CRITICAL";
                riskPriority = 1;
                riskBadgeClass = "bg-rose-500/20 text-rose-300 border-rose-500/40";
            } else if (product.stock < expectedDemandStdLead) {
                riskLevel = "HIGH";
                riskPriority = 2;
                riskBadgeClass = "bg-amber-500/20 text-amber-300 border-amber-500/40";
            } else if (product.stock < (expectedDemandStdLead + safetyBufferUnits)) {
                riskLevel = "MEDIUM";
                riskPriority = 3;
                riskBadgeClass = "bg-sky-500/20 text-sky-300 border-sky-500/40";
            }

            // Evaluate supplier-specific reorder calculations
            const evaluateForSupplier = (sup) => {
                const leadDemand = Math.round(avgDailyDemand * sup.deliveryDays);
                const shortfall = Math.max(0, leadDemand - product.stock);
                // Reorder target covers lead-time demand + safety buffer minus current stock
                const netNeeded = Math.max(0, (leadDemand + safetyBufferUnits) - product.stock);
                const recommendedQty = riskLevel === "LOW" ? 0 : Math.max(sup.moq || 0, netNeeded);
                const totalCost = +(recommendedQty * sup.unitCost).toFixed(2);
                const stockoutGapDays = Math.max(0, +(sup.deliveryDays - daysUntilStockout).toFixed(1));
                const unfulfilledDemandUnits = Math.round(stockoutGapDays * avgDailyDemand);

                return {
                    supplier: sup,
                    leadDemand,
                    shortfall,
                    safetyBufferUnits,
                    recommendedQty,
                    totalCost,
                    stockoutGapDays,
                    unfulfilledDemandUnits
                };
            };

            const speedPlan = evaluateForSupplier(fastestSupplier);
            const costPlan = evaluateForSupplier(cheapestSupplier);
            const allSupplierPlans = product.suppliers.map(evaluateForSupplier);

            return {
                ...product,
                sum7d,
                avgDailyDemand,
                forecastDemand,
                stdLeadDays,
                expectedDemandStdLead,
                daysUntilStockout,
                safetyBufferUnits,
                stdShortfall,
                riskLevel,
                riskPriority,
                riskBadgeClass,
                fastestSupplier,
                cheapestSupplier,
                speedPlan,
                costPlan,
                allSupplierPlans,
                activePlan: preferredStrategy === "speed" ? speedPlan : costPlan
            };
        }

            const barWidth = 34;
            const gap = 18;
            const chartHeight = 110;
            const avgY = chartHeight - (item.avgDailyDemand / maxSale) * (chartHeight - 25);

            const barsSvg = item.salesHistory.map((val, idx) => {
                const h = Math.max(8, (val / maxSale) * (chartHeight - 25));

                    <!-- Primary Supplier Delivery Line -->
                    <line x1="${stdLeadX}" y1="12" x2="${stdLeadX}" y2="${h}" stroke="#f59e0b" stroke-width="1.5" stroke-dasharray="4 2"/>
                    <text x="${stdLeadX}" y="10" text-anchor="middle" fill="#fbbf24" font-size="8.5" font-family="JetBrains Mono">Std Lead (${item.stdLeadDays}d)</text>

                    <!-- Depletion Polyline -->
                    <polyline fill="none" stroke="#f43f5e" stroke-width="2.5" points="${points.join(' ')}"/>

                    <circle cx="${xForDay(0)}" cy="${yForUnits(item.stock)}" r="4" fill="#a3e635"/>

        function dispatchSkuOrder(skuCode) {
            const prod = products.find(p => p.sku === skuCode);
            if (!prod) return;
            const analysis = analyzeProduct(prod);
            const qty = analysis.activePlan.recommendedQty;
            if (qty <= 0) {
                showToast("No Order Required", `${skuCode} already has sufficient stock.`);
                return;
            }
            prod.stock += qty;
            renderAll();
            showToast("Purchase Order Dispatched", `Ordered ${qty} units of ${skuCode} from ${analysis.activePlan.supplier.name}. Inventory replenished.`);
        }

        function executeSinglePurchaseOrder() {
            dispatchSkuOrder(selectedSku);
        }

        function approveAllPurchaseOrders() {
            let count = 0;
            products.forEach(p => {
                const a = analyzeProduct(p);
                if (a.riskLevel !== "LOW" && a.activePlan.recommendedQty > 0) {
                    p.stock += a.activePlan.recommendedQty;
                    count++;
                }
            });
            renderAll();
            showToast("Portfolio Replenished", `Dispatched ${count} automated purchase orders across all at-risk SKUs.`);
        }

            const a = analyzeProduct(p);
            document.getElementById("form-mode").value = "edit";
            document.getElementById("sku-modal-title").innerText = `Edit Telemetry: ${p.sku}`;
            document.getElementById("form-sku-code").value = p.sku;
            document.getElementById("form-sku-code").disabled = true;
            document.getElementById("form-sku-name").value = p.name;
            document.getElementById("form-sku-stock").value = p.stock;
            document.getElementById("form-sku-history").value = p.salesHistory.join(", ");
            document.getElementById("form-fast-cost").value = a.fastestSupplier.unitCost;
            document.getElementById("form-fast-days").value = a.fastestSupplier.deliveryDays;
            document.getElementById("form-cheap-cost").value = a.cheapestSupplier.unitCost;
            document.getElementById("form-cheap-days").value = a.cheapestSupplier.deliveryDays;
            document.getElementById("sku-modal").classList.remove("hidden");
        }

        function closeAddSkuModal() {
            document.getElementById("sku-modal").classList.add("hidden");
        }

        function handleSaveSku(e) {
            e.preventDefault();
            const mode = document.getElementById("form-mode").value;
            const code = document.getElementById("form-sku-code").value.trim();
            const name = document.getElementById("form-sku-name").value.trim();
            const stock = parseInt(document.getElementById("form-sku-stock").value) || 0;
            const rawHistory = document.getElementById("form-sku-history").value.split(",").map(n => parseFloat(n.trim())).filter(n => !isNaN(n) && n >= 0);
            const salesHistory = rawHistory.length > 0 ? rawHistory : [10, 10, 10, 10, 10, 10, 10];

            const fastCost = parseFloat(document.getElementById("form-fast-cost").value) || 50;
            const fastDays = parseInt(document.getElementById("form-fast-days").value) || 3;
            const cheapCost = parseFloat(document.getElementById("form-cheap-cost").value) || 40;
            const cheapDays = parseInt(document.getElementById("form-cheap-days").value) || 7;

            const newSuppliers = [
                { id: "SUP-FAST", name: "Priority Air Logistics", role: "Speed-Optimized", unitCost: fastCost, deliveryDays: fastDays, reliability: 99.1, moq: 15 },
                { id: "SUP-CHEAP", name: "Global Maritime Supply", role: "Cost-Optimized", unitCost: cheapCost, deliveryDays: cheapDays, reliability: 94.0, moq: 25 }
            ];

            if (mode === "edit") {
                const existing = products.find(p => p.sku === code);
                if (existing) {
                    existing.name = name;
                    existing.stock = stock;
                    existing.salesHistory = salesHistory;
                    existing.suppliers = newSuppliers;
                }
            } else {
                products.push({
                    sku: code,
                    name,
                    category: "Custom",
                    stock,
                    salesHistory,
                    primaryLeadDays: Math.round((fastDays + cheapDays) / 2),
                    suppliers: newSuppliers
                });
            }

            selectedSku = code;
            closeAddSkuModal();
            renderAll();
            showToast("Telemetry Saved", `Updated inventory and demand model for ${code}.`);
        }

        function exportProcurementPlanCSV() {
            const analyzed = products.map(analyzeProduct);
            const headers = ["SKU", "Product Name", "Risk Level", "Current Stock", "Avg Daily Demand", "Selected Strategy", "Recommended Supplier", "Delivery Days", "Recommended Order Qty", "Unit Cost", "Total Cost"];
            const rows = analyzed.map(i => [
                i.sku,
                `"${i.name}"`,
                i.riskLevel,
                i.stock,
                i.avgDailyDemand,
                preferredStrategy.toUpperCase(),
                `"${i.activePlan.supplier.name}"`,
                i.activePlan.supplier.deliveryDays,
                i.activePlan.recommendedQty,
                i.activePlan.supplier.unitCost,
                i.activePlan.totalCost
            ]);

            const csvContent = "data:text/csv;charset=utf-8," + [headers.join(","), ...rows.map(r => r.join(","))].join("\n");
            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `SupplyChainIQ_Procurement_Plan_${preferredStrategy}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
            showToast("CSV Exported", "Downloaded portfolio procurement plan CSV.");
        }

        function copyProcurementReport() {
            const analyzed = products.map(analyzeProduct);
            const lines = [
                "=== SUPPLYCHAINIQ: AUTONOMOUS PROCUREMENT DECISION REPORT ===",
                `Active Optimization Strategy: ${preferredStrategy.toUpperCase()} | Safety Buffer: ${safetyBufferDays} Days`,
                ""
            ];
            analyzed.forEach(i => {
                lines.push(`[${i.sku}] ${i.name} | Risk: ${i.riskLevel}`);
                lines.push(`  - Stock: ${i.stock}u | Avg Daily Demand: ${i.avgDailyDemand}u/day | Days to Stockout: ${i.daysUntilStockout}d`);
                lines.push(`  - Recommended Order: ${i.activePlan.recommendedQty}u via ${i.activePlan.supplier.name} (${i.activePlan.supplier.deliveryDays}d @ $${i.activePlan.supplier.unitCost}/u = $${i.activePlan.totalCost})`);
                lines.push(`  - Strategy Comparison: Fastest (${i.speedPlan.supplier.deliveryDays}d, $${i.speedPlan.totalCost}) vs Lowest Cost (${i.costPlan.supplier.deliveryDays}d, $${i.costPlan.totalCost})`);
                lines.push("");
            });

            const ta = document.createElement("textarea");
            ta.value = lines.join("\n");
            document.body.appendChild(ta);
            ta.select();
            document.execCommand("copy");
            document.body.removeChild(ta);
            showToast("Executive Report Copied", "Copied full autonomous procurement report to clipboard.");
        }

            }, 3800);
        }
